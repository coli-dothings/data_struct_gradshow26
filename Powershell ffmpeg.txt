Powershell: Rename all files in folder before the FINAL underscore
Get-ChildItem -File | Rename-Item -NewName { $_.Name -replace '.*_(.*)', 'MPA_4_$1' }

Powershell: Remove all spaces in file name
Get-ChildItem -File | Rename-Item -NewName { $_.Name -replace 'Lin,Zoe', 'LinZoe' }

Powershell: lowercase all text
Get-ChildItem -File | Rename-Item -NewName { $_.Name.ToLower() }


Powershell + ffmpeg: Resize images 
Get-ChildItem *.png | ForEach-Object {
    ffmpeg -i $_.FullName -vf "scale=800:-1" ".\Resized_Images\$($_.Name)"
}


Powershell + ffmpeg: Change file to webp

Get-ChildItem *.jpg | ForEach-Object { ffmpeg -i $_.FullName -c:v libwebp -q:v 75 ("$($_.BaseName).webp") }

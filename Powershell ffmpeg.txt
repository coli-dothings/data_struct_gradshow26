Powershell: Rename all files in folder before an underscore
Get-ChildItem -File | Rename-Item -NewName { $_.Name -replace '^[^_]+', 'PFP' }

Powershell: Rename MPT & SPT
$i = 1; Get-ChildItem -File | Sort-Object Name | ForEach-Object { Rename-Item $_.FullName -NewName ("Project_p{0:D2}$($_.Extension)" -f $i++) }


Powershell: Remove all spaces in file name
Get-ChildItem -File | Rename-Item -NewName { $_.Name -replace ' ', '' }

Powershell: lowercase all text
Get-ChildItem -File | Rename-Item -NewName { $_.Name.ToLower() }


Powershell + ffmpeg: Resize images 
Get-ChildItem *.png | ForEach-Object {
    ffmpeg -i $_.FullName -vf "scale=800:-1" ".\Resized_Images\$($_.Name)"
}


Powershell + ffmpeg: Change file to webp
Get-ChildItem *.png | ForEach-Object { ffmpeg -i $_.FullName -c:v libwebp -q:v 75 ("$($_.BaseName).webp") }

Do this for each file format: png, jpg, jpeg
